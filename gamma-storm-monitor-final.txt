// @version=6
// @title Gamma Storm Monitor - Core Symbols  
// @description Real-time gamma condition monitoring for key symbols
// @author Kepler

// === CONFIGURATION ===
gammaThreshold = input.float(-0.15, "Negative Gamma Threshold", minval=-1.0, maxval=0.0, step=0.01, tooltip="More negative = stronger storm")
narrowZoneThreshold = input.float(12.0, "Narrow Zone Threshold (%)", minval=5.0, maxval=25.0, step=0.5, tooltip="Narrow zone = higher volatility risk")
volumeSpikeThreshold = input.float(1.3, "Volume Spike Threshold", minval=1.0, maxval=3.0, step=0.1, tooltip="Volume spike multiplier")

// === STORM INTENSITY CALCULATION ===
// Approximate gamma-like conditions using available TradingView data
volatility = ta.atr(14) / close * 100
volumeRatio = volume / ta.sma(volume, 20)
rsiValue = ta.rsi(close, 14)
vwapDistance = math.abs(close - ta.vwap) / close * 100

// Storm intensity (0-3 scale)
stormIntensity = 0.0

// Negative gamma conditions (high volatility + volume spike + bearish sentiment)
if volatility > 15 and volumeRatio > volumeSpikeThreshold and rsiValue < 45 and vwapDistance > 1.5
    stormIntensity := 3.0  // Major storm
else if volatility > 12 and volumeRatio > 1.2 and rsiValue < 50
    stormIntensity := 2.0  // Moderate storm
else if volatility > 10 and volumeRatio > 1.1
    stormIntensity := 1.0  // Minor storm
else
    stormIntensity := 0.0  // Normal conditions

// === VISUAL INDICATORS ===
// Background coloring for storm conditions
stormColor = stormIntensity >= 3 ? color.new(color.red, 70) : 
             stormIntensity >= 2 ? color.new(color.orange, 75) : 
             stormIntensity >= 1 ? color.new(color.yellow, 80) : 
             color.new(color.green, 90)

// Storm intensity plot
plot(stormIntensity, title="Storm Intensity", color=stormColor, linewidth=3, style=plot.style_columns)

// Storm markers
plotshape(stormIntensity >= 3, title="Major Storm", location=location.top, color=color.red, style=shape.triangledown, size=size.large, text="STORM")
plotshape(stormIntensity >= 2 and stormIntensity < 3, title="Moderate Storm", location=location.top, color=color.orange, style=shape.diamond, size=size.normal, text="ALERT")
plotshape(stormIntensity >= 1 and stormIntensity < 2, title="Minor Storm", location=location.top, color=color.yellow, style=shape.triangleup, size=size.small, text="WATCH")

// === ALERTS ===
// Major storm alert
if stormIntensity >= 3 and stormIntensity[1] < 3
    alert("MAJOR GAMMA STORM: " + syminfo.ticker + " - Volatility: " + str.tostring(volatility, "#.1f") + "%", alert.freq_once_per_bar)

// Moderate storm alert  
if stormIntensity >= 2 and stormIntensity[1] < 2
    alert("MODERATE GAMMA STORM: " + syminfo.ticker + " - Volatility: " + str.tostring(volatility, "#.1f") + "%", alert.freq_once_per_bar)

// === DASHBOARD INFO ===
// Real-time storm information panel
if barstate.islast
    labelText = "Gamma Storm Monitor\n" + 
                "Symbol: " + syminfo.ticker + "\n" +
                "Intensity: " + str.tostring(math.round(stormIntensity)) + "/3\n" +
                "Volatility: " + str.tostring(volatility, "#.1f") + "%\n" +
                "Volume Spike: " + str.tostring(volumeRatio, "#.1f") + "x\n" +
                "RSI: " + str.tostring(rsiValue, "#.1f") + "\n" +
                "Condition: " + 
                (stormIntensity >= 3 ? "MAJOR STORM" : 
                 stormIntensity >= 2 ? "MODERATE STORM" : 
                 stormIntensity >= 1 ? "MINOR STORM" : "NORMAL")
    
    label.new(bar_index, high * 1.02, text=labelText, 
             color=color.black, textcolor=color.white, 
             size=size.normal, style=label.style_label_up)
