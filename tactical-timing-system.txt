// @version=6
indicator('Gamma Storm Tactical Timing System', shorttitle='GSTTS', overlay=true, precision=0, max_lines_count=500, max_labels_count=500, max_bars_back=500)
// @title Gamma Storm Tactical Timing System
// @description Tactical entry/exit timing based on gamma storm conditions
// @author Kepler

// === TACTICAL TIMING CONFIGURATION ===
// Entry timing based on storm intensity
entryWatchThreshold = input.float(0.5, "Watch Entry Threshold", minval=0.0, maxval=3.0, step=0.1, tooltip="Start preparing positions")
entryMinorThreshold = input.float(1.0, "Minor Entry Threshold", minval=0.5, maxval=3.0, step=0.1, tooltip="Begin entry")
entryModerateThreshold = input.float(2.0, "Moderate Entry Threshold", minval=1.0, maxval=3.0, step=0.1, tooltip="Scale up")
entryMajorThreshold = input.float(3.0, "Major Entry Threshold", minval=2.0, maxval=3.0, step=0.1, tooltip="Full position")

// Risk management based on storm intensity
stopLossTightThreshold = input.float(2.0, "Tight Stop Loss (Storm)", minval=1.0, maxval=3.0, step=0.1, tooltip="Tighten stops during storms")
positionSizeMultiplier = input.float(1.0, "Position Size Multiplier", minval=0.1, maxval=3.0, step=0.1, tooltip="Scale position size with storm intensity")

// === STORM DETECTION (Same as your calibrated system) ===
volatility = ta.atr(14) / close * 100
volumeRatio = volume / ta.sma(volume, 20)
rsiValue = ta.rsi(close, 14)
vwapDistance = math.abs(close - ta.vwap) / close * 100

// Calibrated storm intensity (0-3 scale)
stormIntensity = 0.0
if volatility > 4.0 and volumeRatio > 1.2
    stormIntensity := 3.0  // Major storm
else if volatility > 2.5 and volumeRatio > 1.15  
    stormIntensity := 2.0  // Moderate storm
else if volatility > 1.5 and volumeRatio > 1.1
    stormIntensity := 1.0  // Minor storm
else if volatility > 1.0
    stormIntensity := 0.5  // Watch condition
else
    stormIntensity := 0.0  // Calm conditions

// === TACTICAL TIMING SIGNALS ===
// Entry signals based on storm intensity
entrySignal = 0.0
if stormIntensity >= entryMajorThreshold
    entrySignal := 3.0  // Full position signal
else if stormIntensity >= entryModerateThreshold
    entrySignal := 2.0  // Scale up signal
else if stormIntensity >= entryMinorThreshold
    entrySignal := 1.0  // Initial entry signal
else if stormIntensity >= entryWatchThreshold
    entrySignal := 0.5  // Watch/prepare signal
else
    entrySignal := 0.0  // No entry signal

// Risk management signals based on storm intensity
riskSignal = 0.0
if stormIntensity >= stopLossTightThreshold
    riskSignal := 2.0  // Tight risk management
else if stormIntensity >= 1.0
    riskSignal := 1.0  // Standard risk management
else
    riskSignal := 0.0  // Normal risk management

// === TACTICAL INDICATORS ===
// Entry timing visualization
plot(entrySignal, title="Entry Signal Strength", color=color.new(color.green, 0), linewidth=4, style=plot.style_columns)
plot(riskSignal, title="Risk Management Level", color=color.new(color.orange, 0), linewidth=3, style=plot.style_area)

// Entry timing markers
plotshape(entrySignal >= 3, title="Full Position Signal", location=location.top, color=color.green, style=shape.triangleup, size=size.large, text="FULL")
plotshape(entrySignal >= 2 and entrySignal < 3, title="Scale Up Signal", location=location.top, color=color.lime, style=shape.triangleup, size=size.normal, text="SCALE")
plotshape(entrySignal >= 1 and entrySignal < 2, title="Entry Signal", location=location.top, color=color.yellow, style=shape.triangleup, size=size.normal, text="ENTER")
plotshape(entrySignal >= 0.5 and entrySignal < 1, title="Watch Signal", location=location.top, color=color.blue, style=shape.circle, size=size.small, text="WATCH")

// Risk management indicators
plotshape(riskSignal >= 2, title="Tight Risk Management", location=location.bottom, color=color.red, style=shape.diamond, size=size.normal, text="TIGHT")
plotshape(riskSignal >= 1 and riskSignal < 2, title="Standard Risk Management", location=location.bottom, color=color.orange, style=shape.diamond, size=size.small, text="STANDARD")

// === TACTICAL ALERTS ===
// Entry timing alerts
if entrySignal >= 3 and entrySignal[1] < 3
    alert("FULL POSITION SIGNAL: " + syminfo.ticker + " - Storm: " + str.tostring(stormIntensity, "#.1f"), alert.freq_once_per_bar)

if entrySignal >= 2 and entrySignal[1] < 2
    alert("SCALE UP SIGNAL: " + syminfo.ticker + " - Storm: " + str.tostring(stormIntensity, "#.1f"), alert.freq_once_per_bar)

if entrySignal >= 1 and entrySignal[1] < 1
    alert("ENTRY SIGNAL: " + syminfo.ticker + " - Storm: " + str.tostring(stormIntensity, "#.1f"), alert.freq_once_per_bar)

if entrySignal >= 0.5 and entrySignal[1] < 0.5
    alert("WATCH SIGNAL: " + syminfo.ticker + " - Storm: " + str.tostring(stormIntensity, "#.1f"), alert.freq_once_per_bar)

// === TACTICAL DASHBOARD ===
// Tactical timing information
if barstate.islast
    tacticalText = "TACTICAL TIMING SYSTEM\n"
    tacticalText := tacticalText + "Symbol: " + syminfo.ticker + "\n"
    tacticalText := tacticalText + "Storm Intensity: " + str.tostring(stormIntensity, "#.1f") + "/3\n"
    tacticalText := tacticalText + "Entry Signal: " + str.tostring(entrySignal, "#.1f") + "/3\n"
    tacticalText := tacticalText + "Risk Level: " + str.tostring(riskSignal, "#.1f") + "/2\n"
    
    if entrySignal >= 3
        tacticalText := tacticalText + "Action: FULL POSITION"
    else if entrySignal >= 2
        tacticalText := tacticalText + "Action: SCALE UP"
    else if entrySignal >= 1
        tacticalText := tacticalText + "Action: ENTER POSITION"
    else if entrySignal >= 0.5
        tacticalText := tacticalText + "Action: PREPARE/WATCH"
    else
        tacticalText := tacticalText + "Action: WAIT/NO ENTRY"
    
    if riskSignal >= 2
        tacticalText := tacticalText + "\nRisk: TIGHT MANAGEMENT"
    else if riskSignal >= 1
        tacticalText := tacticalText + "\nRisk: STANDARD MANAGEMENT"
    else
        tacticalText := tacticalText + "\nRisk: NORMAL MANAGEMENT"
    
    label.new(bar_index, low * 0.98, text=tacticalText, 
             color=color.black, textcolor=color.white, 
             size=size.normal, style=label.style_label_down)
