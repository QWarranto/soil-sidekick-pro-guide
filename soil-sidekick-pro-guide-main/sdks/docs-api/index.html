    LeafEngines SDK v2.0

    Official TypeScript/JavaScript SDK for LeafEngines API - Agricultural intelligence platform with Skyline Instruments sensor integration.

    Installation

    npm install @soilsidekick/sdk

    Quick Start

    import { LeafEnginesClient } from '@soilsidekick/sdk';

    const client = new LeafEnginesClient({
      apiKey: 'sk_live_your_api_key',
      projectRef: 'your-project-ref',
    });

    // Register a Skyline sensor device
    const device = await client.sensors.devices.register({
      deviceId: 'skyline-mmWave-001',
      deviceType: 'mmwave_radar',
      farmId: 'farm-123',
      firmwareVersion: '1.2.0',
    });

    console.log(`Device registered: ${device.deviceId}`);

    Sensor Integration

    Register Devices

    // Register new sensor
    const device = await client.sensors.devices.register({
      deviceId: 'skyline-001',
      deviceType: 'mmwave_radar',  // or 'quantum_rf', 'precision_timing'
      farmId: 'farm-123',
      firmwareVersion: '1.0.0',
    });

    // List all devices
    const devices = await client.sensors.devices.list({
      farmId: 'farm-123',
      status: 'active',
    });

    // Get device details
    const details = await client.sensors.devices.get('skyline-001');

    // Update device
    await client.sensors.devices.update('skyline-001', {
      firmwareVersion: '1.1.0',
    });

    Send Sensor Data

    // Send readings via HTTP (MQTT fallback)
    await client.sensors.readings.send({
      deviceId: 'skyline-001',
      deviceType: 'mmwave_radar',
      timestamp: new Date(),
      readings: [
        {
          metric: 'reflectivity',
          value: 0.92,
          unit: 'ratio',
          confidence: 0.94,
          timestamp: new Date(),
        },
        {
          metric: 'canopy_density',
          value: 0.78,
          unit: 'ratio',
          timestamp: new Date(),
        },
      ],
      metadata: {
        firmwareVersion: '1.0.0',
        batteryLevel: 95,
        signalStrength: -45,
      },
    });

    Query Historical Data

    // Get readings for time range
    const readings = await client.sensors.readings.query({
      deviceId: 'skyline-001',
      metric: 'reflectivity',
      startTime: new Date('2026-02-01'),
      endTime: new Date(),
      limit: 1000,
    });

    // Get latest reading
    const latest = await client.sensors.readings.getLatest('skyline-001', 'reflectivity');

    // Get aggregated data (hourly averages)
    const hourly = await client.sensors.readings.aggregate({
      deviceId: 'skyline-001',
      metric: 'reflectivity',
      startTime: new Date('2026-02-01'),
      endTime: new Date(),
      interval: '1h',
    });

    Real-time Streaming

    // Subscribe to real-time readings
    const subscription = client.sensors.readings.subscribe(
      'skyline-001',
      (reading) => {
        console.log(`New reading: ${reading.metric} = ${reading.value}`);
      }
    );

    // Later: unsubscribe
    subscription.unsubscribe();

    Alert Management

    // Subscribe to critical alerts
    client.sensors.alerts.subscribe(
      (alert) => {
        if (alert.severity === 'critical') {
          sendEmergencyNotification(alert);
        }
      },
      { severity: ['warning', 'critical'] }
    );

    // Get unacknowledged alerts
    const alerts = await client.sensors.alerts.getAlerts({
      acknowledged: false,
      severity: 'critical',
    });

    // Acknowledge alert
    await client.sensors.alerts.acknowledge(alerts[0].id);

    MQTT Direct Connection

    For high-frequency data publishing, connect directly to MQTT broker:

    // Connect to MQTT broker
    await client.connectMQTT({
      host: 'your-cluster.s1.eu.hivemq.cloud',
      port: 8883,
      protocol: 'mqtts',
      username: 'your-username',
      password: 'your-password',
    });

    // Publish sensor data
    client.mqtt!.publish(
      'skyline/skyline-001/readings',
      {
        device_id: 'skyline-001',
        device_type: 'mmwave_radar',
        timestamp: new Date().toISOString(),
        readings: [
          { metric: 'reflectivity', value: 0.92, unit: 'ratio' }
        ],
        metadata: { firmware_version: '1.0.0' },
      }
    );

    // Subscribe to topics
    client.mqtt!.subscribe('skyline/+/status', (message) => {
      console.log('Device status:', message);
    });

    // Monitor connection
    client.mqtt!.onStatusChange((status) => {
      console.log('MQTT status:', status);
    });

    // Disconnect when done
    client.disconnectMQTT();

    Error Handling

    import { LeafEnginesError } from '@soilsidekick/sdk';

    try {
      await client.sensors.devices.get('unknown-device');
    } catch (error) {
      if (error instanceof LeafEnginesError) {
        console.error(`API Error ${error.code}: ${error.message}`);
        console.error('Details:', error.details);
      }
    }
